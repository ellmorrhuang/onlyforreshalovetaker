    function addNumber(num){
      const input = document.getElementById('codeInput');

      if(input.value.length === 2 || input.value.length === 5){
        input.value += '-';
      }

      if(input.value.length < 8){
        input.value += num;
      }
    }

    function removeNumber(){
      const input = document.getElementById('codeInput');
      input.value = input.value.slice(0,-1);
    }

    let currentQuestion = 0;

    function loadQuestion(){
      const q = questions[currentQuestion];

      document.getElementById('question').innerText = q.question;
      document.getElementById('quizMessage').innerText = '';
      document.getElementById('correctMessage').innerText = ''; 

      const optionsDiv = document.getElementById('options');
      optionsDiv.innerHTML = '';

      q.options.forEach(option => {
        const btn = document.createElement('button');
        btn.classList.add('option');
        btn.innerText = option;

        btn.onclick = () => {
          if(option === q.answer){
            document.getElementById('correctMessage').innerText = q.correct;

            setTimeout(() => {
              currentQuestion++;

              if(currentQuestion < questions.length){
                loadQuestion();
              } else {
                document.getElementById('quizPage').classList.add('hidden');
                document.getElementById('rewardPage').classList.remove('hidden');
              }
            }, 1200);
          } else {
              document.getElementById('quizPage').classList.add('hidden');
              document.getElementById('finalPage').classList.remove('hidden');
            }
          } else {
            document.getElementById('quizMessage').innerText = q.wrong;
            document.getElementById('quizPage').classList.add('shake');

            setTimeout(() => {
              document.getElementById('quizPage').classList.remove('shake');

        }
